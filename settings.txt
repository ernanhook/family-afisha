# Основные настройки. Формат: ключ = значение. Строки с # — комментарии.

# Кто идёт (ИИ учитывает это при отборе)
child = ребёнок 1,5–2 лет
# Дата рождения ребёнка (2025-08-01, 01.08.2025 или «1 августа 2025»). Бот сам считает
# возраст и «растёт» вместе с ребёнком: через год события 3+ начнут попадать в сводку.
# Тогда строка child выше не нужна.
child_birth = 2025-08-01
family = семья из трёх человек
home_city = Рига

# Регион поиска: вся Латвия / Рига и окрестности / Курземе и т.п.
region = вся Латвия

# Когда присылать сводку: день недели (пн, вт, ср, чт, пт, сб, вс) и час по Риге
# (день можно писать полностью: четверг)
digest_day = чт
digest_hour = 10

# На сколько дней вперёд искать события (14 = две недели, считая с сегодняшнего дня)
days_ahead = 14

# Не показывать события, которые начинаются позже этого часа (19 = до 19:00)
latest_start_hour = 19

# Город для прогноза погоды: широта и долгота (Рига: 56.95, 24.11)
weather_lat = 56.95
weather_lon = 24.11
weather_name = Рига

# Сколько постоянных мест предлагать (0 — выключить блок)
places_count = 3

# Через сколько дней место из «Куда ещё сходить» можно предложить снова
places_repeat_days = 28

# За сколько дней напоминать о праздниках в блоке «Скоро» (0 — выключить блок)
holidays_ahead_days = 21

# Лимит расходов на ИИ в месяц, $ — это «предохранитель». Обычный месяц стоит

monthly_budget_usd = 1.50

# Строка диагностики в конце сводки: auto (если мало событий или ошибки), always, never
show_diagnostics = auto

# Сколько запросов к Google Events (OpenWeb Ninja) можно сделать за месяц.
# Бесплатно — 50; при 5 запросах в неделю это около 22 в месяц.
events_api_monthly_limit = 45

# Модель ИИ для поиска событий (дешёвая, читает много текста)
model = claude-haiku-5-5

# Модель для вычитки русского текста карточек перед отправкой: исправляет ошибки,
# недопереводы и смесь латиницы с кириллицей. Около 3–4 центов за сводку. Выключить: no
proofread_model = claude-sonnet-5-5

# Совет недели в конце сводки: ИИ выбирает самое интересное событие на каждую неделю
# (yes — показывать, no — не показывать). Стоит около 0,1 цента за сводку.
week_tips = yes

# Напоминание на выходные: день и час (пт 18 = в пятницу в 18:00). Короткий список
# событий на субботу и воскресенье из последней сводки и свежая погода. Выключить: no
weekend_reminder = пт 18

# Срочные анонсы: если на новостных порталах объявили что-то крупное (цирк, большой
# семейный фестиваль), бот сразу пришлёт карточку, не дожидаясь сводки.
# Сколько таких сообщений максимум в неделю (0 — выключить)
urgent_per_week = 2

# Больше скольких карточек в сводке не присылать (остальное — коротким списком)
max_cards = 30

# Календарь: apple — для iPhone: бот ведёт календарь «Мы идём» (только события, на которые
# поставили 👍) и один раз пришлёт ссылку, как подключить его на iPhone;
# google — под каждой карточкой кнопка «📅 В календарь» для Google Календаря
calendar = apple
